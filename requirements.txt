# Netwise Python dependencies
#
# Install with:  pip install -r requirements.txt

# --- Layer 1: Batfish orchestration ----------------------------------------
# Pinned deliberately. The pybatfish client and the Batfish service in the
# Docker container should be on matching versions -- skew between them causes
# confusing failures. This pin matches image batfish/allinone reporting
# Batfish 2025.07.07.2423. If you pull a newer image, bump this to match.
pybatfish==2025.7.7.2423

# Batfish answers come back as pandas DataFrames, and our analysis code works
# with them directly, so pandas is a direct dependency rather than an
# incidental one.
pandas>=2.0

# --- Layer 2: AI explanation layer -----------------------------------------
# TODO: add the Ollama client once the AI layer starts.

# --- Layer 3: Web interface ------------------------------------------------
# Chosen over Flask for the auto-generated API docs at /docs (useful evidence
# in a review) and native async, which the streaming chat pane will need.
fastapi>=0.110
uvicorn[standard]>=0.27

# FastAPI needs this to parse multipart form data -- i.e. file uploads. Without
# it, the config upload endpoint fails at import time rather than at request
# time, so it is a hard dependency, not an optional extra.
python-multipart>=0.0.9
